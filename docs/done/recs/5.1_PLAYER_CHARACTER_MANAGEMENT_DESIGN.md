### 1.2 Shared Patterns

- **Concurrency model:** Last-write-wins. No optimistic locking.
- **Audit logging:** All mutations call `log_admin_action(db, admin_email, action, target_type, target_id, details, ip)` — the existing helper in `admin_players.py`.
- **Stat recalculation:** Any mutation that affects stat sources (class, level, equipment, skills, artifacts) calls `recalculate_character_stats(character_id, db)` before returning.
- **Admin auth:** All endpoints use `Depends(get_current_admin)` which verifies `is_system_admin or is_game_admin`.
- **Error responses:** 422 for validation, 404 for not found, 409 for constraint violations. All include `{"detail": "Human-readable message"}`.

```python
async def edit_character(db, character_id, payload, admin_email, ip):
    character = db.get(PlayerCharacter, character_id)
    if not character: raise 404

    old_state = snapshot(character)  # capture before values
    changes = {}

    # Simple field updates
    if payload.character_name is not None:
        validate_name(payload.character_name)  # length + profanity
        character.character_name = payload.character_name
        changes["character_name"] = (old_state.character_name, payload.character_name)

    if payload.level is not None:
        character.level = clamp(payload.level, 1, 999)
        changes["level"] = (old_state.level, character.level)

    if payload.character_xp is not None:
        character.character_xp = max(0, payload.character_xp)
        changes["character_xp"] = (old_state.character_xp, character.character_xp)

    # Class reassignment (complex flow)
    if payload.class_id is not None and payload.class_id != character.class_id:
        skill_mapping = await reassign_class(db, character, payload.class_id)
        changes["class_reassignment"] = skill_mapping

    recalculate_character_stats(character.id, db)
    log_admin_action(db, admin_email, "character_edited", "character", character_id,
                     {"changes": changes, "old_state": old_state}, ip)
    db.commit()
    return character_with_stats(db, character)
```

### 2.2 Class Reassignment Logic

```python
async def reassign_class(db, character, new_class_id):
    new_class = db.get(CharacterClass, new_class_id)
    if not new_class: raise 404("Target class not found")

    old_class_id = character.class_id

    # Get skills categorized
    all_skills = db.exec(select(Skill)).all()
    old_exclusive = sorted(
        [s for s in all_skills if s.class_id == old_class_id and s.is_class_exclusive],
        key=lambda s: s.id  # sort by creation order (Skill has no sort_order)
    )
    new_exclusive = sorted(
        [s for s in all_skills if s.class_id == new_class_id and s.is_class_exclusive],
        key=lambda s: s.id  # sort by creation order (Skill has no sort_order)
    )

    # Get current skill levels for old exclusive skills
    old_skill_levels = {
        csl.skill_id: (csl.level, csl.current_xp)
        for csl in db.exec(
            select(CharacterSkillLevel)
            .where(CharacterSkillLevel.character_id == character.id)
            .where(CharacterSkillLevel.skill_id.in_([s.id for s in old_exclusive]))
        ).all()
    }

    # Map by sort order: 1st old → 1st new, 2nd old → 2nd new
    mapping = []
    for i, new_skill in enumerate(new_exclusive):
        if i < len(old_exclusive):
            old_skill = old_exclusive[i]
            old_level, old_xp = old_skill_levels.get(old_skill.id, (1, 0))

            # Delete old skill level record
            old_csl = db.exec(
                select(CharacterSkillLevel)
                .where(CharacterSkillLevel.character_id == character.id)
                .where(CharacterSkillLevel.skill_id == old_skill.id)
            ).first()
            if old_csl:
                db.delete(old_csl)

            # Create new skill level record with transferred levels
            new_csl = CharacterSkillLevel(
                character_id=character.id,
                skill_id=new_skill.id,
                level=old_level,
                current_xp=old_xp
            )
            db.add(new_csl)
            mapping.append({
                "old_skill_id": old_skill.id, "old_skill_name": old_skill.name,
                "new_skill_id": new_skill.id, "new_skill_name": new_skill.name,
                "level_transferred": old_level, "xp_transferred": old_xp
            })
        else:
            # Extra new skill — start at level 1
            new_csl = CharacterSkillLevel(
                character_id=character.id,
                skill_id=new_skill.id,
                level=1, current_xp=0
            )
            db.add(new_csl)
            mapping.append({
                "old_skill_id": None, "old_skill_name": None,
                "new_skill_id": new_skill.id, "new_skill_name": new_skill.name,
                "level_transferred": 1, "xp_transferred": 0
            })

    # Excess old skills — levels lost
    lost = []
    for i in range(len(new_exclusive), len(old_exclusive)):
        old_skill = old_exclusive[i]
        old_level, old_xp = old_skill_levels.get(old_skill.id, (1, 0))
        old_csl = db.exec(
            select(CharacterSkillLevel)
            .where(CharacterSkillLevel.character_id == character.id)
            .where(CharacterSkillLevel.skill_id == old_skill.id)
        ).first()
        if old_csl:
            db.delete(old_csl)
        lost.append({
            "skill_id": old_skill.id, "skill_name": old_skill.name,
            "level_lost": old_level
        })

    character.class_id = new_class_id

    # Check equipment requirements post-reassignment
    equipment_warnings = check_equipment_requirements(db, character)

    return {
        "old_class_id": old_class_id, "new_class_id": new_class_id,
        "skill_mapping": mapping, "skills_lost": lost,
        "equipment_warnings": equipment_warnings
    }
```

**Service Logic:**

```python
async def craft_item(db, character_id, payload, admin_email, ip):
    character = db.get(PlayerCharacter, character_id)
    if not character: raise 404

    if payload.mode == "manual":
        item = craft_item_manual(db, character, payload)
    else:
        item = craft_item_random(db, character, payload)

    # Add to inventory (bag)
    bag_count = count_bag_items(db, character.id)
    inventory_entry = PlayerInventory(
        character_id=character.id,
        item_id=item.id,
        is_equipped=False,
        equipped_slot=None,
        quantity=1
    )
    db.add(inventory_entry)

    log_admin_action(db, admin_email, "item_crafted", "character", character_id, {
        "item_id": item.id, "item_name": item.name, "rarity": item.rarity,
        "mode": payload.mode, "bag_count": bag_count + 1,
        "exceeded_cap": bag_count >= 10
    }, ip)
    db.commit()

    return {
        "item": serialize_item(item),
        "inventory_id": inventory_entry.id,
        "bag_warning": bag_count >= 10
    }


def craft_item_manual(db, character, payload):
    type_base = db.get(ItemTypeBase, payload.type_base_id)
    prefix = db.get(ItemPrefix, payload.prefix_id)
    quality = db.get(ItemQuality, payload.quality_id)
    lore_tag = db.get(ItemLoreTag, payload.lore_tag_id)
    suffix = db.get(ItemSuffix, payload.suffix_id)

    # Validate base_stat_values within range
    for stat, value in payload.base_stat_values.items():
        range_min, range_max = type_base.base_stat_range.get(stat, [0, 0])
        if not (range_min <= value <= range_max):
            raise 422(f"{stat} value {value} outside range [{range_min}, {range_max}]")

    # Calculate final stats
    rarity_mult = RARITY_MULTIPLIERS[payload.rarity]
    base_stats = {}
    for stat, base_val in payload.base_stat_values.items():
        prefix_bonus = prefix.stat_bonuses.get(stat, 0)
        suffix_bonus = suffix.stat_bonuses.get(stat, 0)
        base_stats[stat] = math.floor((base_val + prefix_bonus + suffix_bonus) * rarity_mult)

    # Build item code and name
    item_code = f"{prefix.code}_{quality.code}_{lore_tag.code}_{type_base.code}_{suffix.code}"
    name = f"{prefix.display_name} {quality.display_name} {lore_tag.display_name} {type_base.display_name} {suffix.display_name}"

    # Level and requirements
    item_level = payload.item_level or character.level
    req_fraction = {"common": 0.3, "uncommon": 0.35, "rare": 0.40, "epic": 0.45, "cosmic": 0.50}
    stat_requirements = {
        stat: max(1, math.floor(val * req_fraction[payload.rarity]))
        for stat, val in base_stats.items()
    }

    item = InventoryItem(
        name=name, item_type=type_base.display_name, rarity=payload.rarity,
        base_stats=base_stats, item_code=item_code, item_level=item_level,
        min_char_level=max(1, item_level - 2),
        stat_requirements=stat_requirements,
        gear_slot_id=payload.gear_slot_id,
        is_dream_item=True, acquired_from="admin_grant"
    )
    db.add(item)
    db.flush()
    return item
```

### 3.3 Backend: `PATCH /api/admin/items/{item_id}`

**Request Body:**
```json
{
  "prefix_id": 8,
  "quality_id": null,
  "lore_tag_id": null,
  "suffix_id": 12,
  "rarity": "epic"
}
```

Only non-null fields are updated. Stats are recalculated from the (possibly updated) components. The item's `item_code` and `name` are regenerated.

**Service Logic:**
1. Load current item + all its component references (parse `item_code` to find current prefix/quality/lore_tag/type_base/suffix).
2. Replace specified components.
3. Recalculate stats using the same formula.
4. If item is equipped → call `recalculate_character_stats()`.
5. Log before/after diff to audit log.

### 3.4 Backend: `PATCH /api/admin/artifacts/{artifact_id}`

**Request Body:**
```json
{
  "rarity": "epic",
  "prefix_code": "blazing",
  "suffix_code": "of_echoes"
}
```

**Service Logic:**

For **generated artifacts** (`artifact_type = "generated"`):
1. Parse existing `artifact_code` (format: `{prefix}_{type}_{suffix}`) to identify current components.
2. Look up new prefix/suffix from `ArtifactPrefix`/`ArtifactSuffix` tables by code.
3. Retain the type_code from the parsed artifact_code (type base is not editable).
4. Recalculate `stat_bonuses` using artifact generation formula (base_stat_range from type × rarity multiplier + prefix/suffix bonuses).
5. Regenerate `artifact_code` = `{prefix_code}_{type_code}_{suffix_code}`.
6. Regenerate `name` = `{prefix.display_name} {type.display_name} {suffix.display_name}`.
7. Update `PlayerArtifact`.

For **curated artifacts** (`artifact_type = "curated"`):
1. Only `rarity` is editable.
2. Look up the `CuratedArtifactTier` row matching `(curated_artifact_id, rarity)`.
3. If no tier exists for that rarity → 422 error.
4. Update `PlayerArtifact.rarity`, `stat_bonuses` from tier, `unique_effect_text` from tier.

Both paths: call `recalculate_character_stats()`, log to audit.

**Service Logic:**
```python
async def adjust_essence(db, character_id, amount, direction, reason, admin_email, ip):
    character = db.get(PlayerCharacter, character_id)
    essence = db.exec(
        select(PlayerEssence).where(PlayerEssence.character_id == character_id)
    ).first()
    meta = db.exec(
        select(PlayerMetaProgression).where(PlayerMetaProgression.player_id == character.player_id)
    ).first()

    old_balance = essence.current_balance

    if direction == "grant":
        essence.current_balance += amount
        meta.elysium_essence += amount
        meta.total_essence_earned += amount
    else:  # debit
        if amount > essence.current_balance:
            raise 422(f"Debit {amount} exceeds balance {essence.current_balance}")
        essence.current_balance -= amount
        meta.spent_essence += amount

    # Create adjustment record
    adjustment = AdminEssenceAdjustment(
        character_id=character_id,
        player_id=character.player_id,
        admin_email=admin_email,
        adjustment_type=direction,
        amount=amount,
        balance_before=old_balance,
        balance_after=essence.current_balance,
        reason=reason
    )
    db.add(adjustment)

    log_admin_action(db, admin_email, f"essence_{direction}ed", "character", character_id, {
        "amount": amount, "direction": direction,
        "balance_before": old_balance, "balance_after": essence.current_balance,
        "reason": reason
    }, ip)
    db.commit()

    return {
        "balance_before": old_balance,
        "balance_after": essence.current_balance,
        "adjustment_id": adjustment.id
    }
```

```python
async def set_progression(db, character_id, target_book, target_chapter, target_scene, admin_email, ip):
    character = db.get(PlayerCharacter, character_id)
    progress = db.exec(
        select(PlayerProgress).where(PlayerProgress.character_id == character_id)
    ).first()

    old_position = (progress.book_number, progress.chapter_number, progress.scene_number)
    new_position = (target_book, target_chapter, target_scene)

    # Validate target exists
    target_scene_obj = db.exec(
        select(Scene).join(Chapter).join(Book)
        .where(Book.book_number == target_book)
        .where(Chapter.chapter_number == target_chapter)
        .where(Scene.scene_number == target_scene)
    ).first()
    if not target_scene_obj: raise 404("Target scene not found")

    direction = "forward" if new_position > old_position else "backward"

    if direction == "forward":
        # Backfill all prior scenes
        prior_scenes = db.exec(
            select(Scene).join(Chapter).join(Book)
            .where(
                (Book.book_number < target_book) |
                ((Book.book_number == target_book) & (Chapter.chapter_number < target_chapter)) |
                ((Book.book_number == target_book) & (Chapter.chapter_number == target_chapter) & (Scene.scene_number < target_scene))
            )
            .order_by(Book.book_number, Chapter.chapter_number, Scene.scene_number)
        ).all()

        scenes_backfilled = 0
        bosses_backfilled = 0
        for scene in prior_scenes:
            # Create PlayerSceneRecord if missing (uses player_id, not character_id)
            existing = db.exec(
                select(PlayerSceneRecord)
                .where(PlayerSceneRecord.player_id == character.player_id)
                .where(PlayerSceneRecord.scene_id == scene.id)
            ).first()
            if not existing:
                record = PlayerSceneRecord(
                    player_id=character.player_id,
                    scene_id=scene.id,
                    first_completed_at=datetime.utcnow(),
                    best_wave=1, best_time_seconds=0,
                    total_enemies_killed=0, total_runs=1
                )
                db.add(record)
                scenes_backfilled += 1

            # Create boss_completion if boss scene and missing
            if scene.scene_type in ("chapter_boss", "book_boss"):
                existing_boss = db.exec(
                    select(BossCompletion)
                    .where(BossCompletion.player_id == character.player_id)
                    .where(BossCompletion.scene_id == scene.id)
                ).first()
                if not existing_boss:
                    boss_record = BossCompletion(
                        player_id=character.player_id,
                        scene_id=scene.id,
                        boss_type=scene.scene_type,
                        chapter_id=scene.chapter_id,
                        book_id=scene.chapter.book_id if scene.scene_type == "book_boss" else None,
                        completed_at=datetime.utcnow()
                    )
                    db.add(boss_record)
                    bosses_backfilled += 1

    # Update progress
    progress.book_number = target_book
    progress.chapter_number = target_chapter
    progress.scene_number = target_scene

    log_admin_action(db, admin_email, "progression_set", "character", character_id, {
        "old_position": old_position, "new_position": new_position,
        "direction": direction,
        "scenes_backfilled": scenes_backfilled if direction == "forward" else 0,
        "bosses_backfilled": bosses_backfilled if direction == "forward" else 0
    }, ip)
    db.commit()

    return {
        "old_position": { "book": old_position[0], "chapter": old_position[1], "scene": old_position[2] },
        "new_position": { "book": target_book, "chapter": target_chapter, "scene": target_scene },
        "direction": direction,
        "scenes_backfilled": scenes_backfilled if direction == "forward" else 0,
        "bosses_backfilled": bosses_backfilled if direction == "forward" else 0
    }
```

```python
async def update_skills(db, character_id, updates, admin_email, ip):
    character = db.get(PlayerCharacter, character_id)
    results = []

    for update in updates:
        skill = db.get(Skill, update.skill_id)
        if not skill: raise 404(f"Skill {update.skill_id} not found")

        # Check prerequisites
        prereqs_met = evaluate_prerequisites(db, character_id, update.skill_id)
        if not prereqs_met:
            raise 422(f"Skill '{skill.name}' has unmet prerequisites")

        # Get or create CharacterSkillLevel
        csl = db.exec(
            select(CharacterSkillLevel)
            .where(CharacterSkillLevel.character_id == character_id)
            .where(CharacterSkillLevel.skill_id == update.skill_id)
        ).first()

        old_level = csl.level if csl else None
        old_xp = csl.current_xp if csl else None

        if csl:
            csl.level = clamp(update.level, 1, 999)
            csl.current_xp = max(0, update.xp)
        else:
            csl = CharacterSkillLevel(
                character_id=character_id,
                skill_id=update.skill_id,
                level=clamp(update.level, 1, 999),
                current_xp=max(0, update.xp)
            )
            db.add(csl)

        results.append({
            "skill_id": update.skill_id, "skill_name": skill.name,
            "old_level": old_level, "new_level": csl.level,
            "old_xp": old_xp, "new_xp": csl.current_xp
        })

    recalculate_character_stats(character_id, db)

    log_admin_action(db, admin_email, "skills_edited", "character", character_id,
                     {"updates": results}, ip)
    db.commit()
    return results
```

## 7. Activity Timeline

### 7.1 Backend: `GET /api/admin/players/{player_id}/timeline`

**Query Params:**
- `categories`: comma-separated filter (e.g., `gameplay,economy,admin`). Default: all.
- `before`: ISO datetime cursor for pagination (load events before this timestamp).
- `limit`: events per page (default 50, max 200).

**Categories and their sources:**

| Category | Sources | Event Types |
|:---|:---|:---|
| `gameplay` | `player_story_sessions`, `player_achievements`, `player_scene_records` | session_start, session_complete, achievement_unlocked |
| `economy` | `payment_orders`, `shard_transactions`, `admin_essence_adjustments` | purchase, subscription_change, shard_transaction, essence_adjustment |
| `admin` | `admin_audit_log` | All admin actions targeting this player |
| `social` | `support_tickets` | ticket_created, ticket_updated |
| `anticheat` | `activity_events` (type=anti_cheat_anomaly) | anti_cheat_anomaly |

**Service Logic:**
```python
async def get_player_timeline(db, player_id, categories, before, limit):
    events = []

    if "gameplay" in categories:
        # Story sessions
        sessions = db.exec(
            select(PlayerStorySession)
            .where(PlayerStorySession.player_id == player_id)
            .where(PlayerStorySession.created_at < before if before else True)
            .order_by(PlayerStorySession.created_at.desc())
            .limit(limit)
        ).all()
        for s in sessions:
            events.append({
                "timestamp": s.created_at,
                "category": "gameplay",
                "type": "session_start",
                "summary": f"Started session on scene '{s.scene.title}'",
                "detail": { "session_id": s.id, "scene_id": s.scene_id }
            })
            if not s.is_active:
                events.append({
                    "timestamp": s.updated_at,
                    "category": "gameplay",
                    "type": "session_complete",
                    "summary": f"Completed scene '{s.scene.title}'",
                    "detail": {
                        "session_id": s.id,
                        "waves_completed": s.current_wave,
                        "session_gold": s.session_gold,
                        "deaths": s.deaths,
                        "dark_ritual_multiplier": s.dark_ritual_multiplier,
                        "duration_seconds": (s.updated_at - s.created_at).total_seconds()
                    }
                })

        # Achievements
        achievements = db.exec(
            select(PlayerAchievement).join(Achievement)
            .where(PlayerAchievement.player_id == player_id)
            .where(PlayerAchievement.earned_at < before if before else True)
            .where(PlayerAchievement.is_completed == True)
            .order_by(PlayerAchievement.earned_at.desc())
            .limit(limit)
        ).all()
        for a in achievements:
            events.append({
                "timestamp": a.earned_at,
                "category": "gameplay",
                "type": "achievement_unlocked",
                "summary": f"Unlocked '{a.achievement.name}'",
                "detail": {
                    "achievement_id": a.achievement_id,
                    "category": a.achievement.category,
                    "rarity": a.achievement.rarity
                }
            })

    if "economy" in categories:
        # ... similar queries for payment_orders, shard_transactions, essence_adjustments

    if "admin" in categories:
        # ... query admin_audit_log where target_id = player_id

    if "social" in categories:
        # ... query support_tickets where player_id = player_id

    if "anticheat" in categories:
        # ... query activity_events where player_id = player_id and type = 'anti_cheat_anomaly'

    # Sort all events by timestamp descending, apply limit
    events.sort(key=lambda e: e["timestamp"], reverse=True)
    events = events[:limit]

    return {
        "events": events,
        "next_cursor": events[-1]["timestamp"].isoformat() if events else None,
        "has_more": len(events) == limit
```

## 10. Design Decisions (Resolved)

| # | Decision | Resolution |
|:---|:---|:---|
| 1 | Stat breakdown detail | Numeric value + expandable contributing source list (items, skills) |
| 2 | Manual mode base stat selection | Range slider with min/max from type base (admin picks exact value) |
| 3 | Item name override | Not allowed — always derived from components |
| 4 | Scene picker format | Names only (no completion stats in dropdown) |
| 5 | Session complete detail in timeline | Full session stats (waves, gold, time, deaths, items) |
| 6 | Artifact editing | In scope — generated (prefix/suffix/rarity) and curated (rarity tier) |
| 7 | Concurrent edit protection | Last-write-wins (no optimistic locking) |
| 8 | Component file for item editor | Single `ItemEditorModal.tsx` handles both items and artifacts (prop-driven) |
| 9 | Timeline aggregation | Server-side union of multiple tables, sorted by timestamp |
