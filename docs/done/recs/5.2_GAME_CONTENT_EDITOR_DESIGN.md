### 1.2 Shared Patterns

- **Concurrency model:** Last-write-wins. No optimistic locking.
- **Audit logging:** All mutations call `log_admin_action(db, admin_email, action, target_type, target_id, details, ip)` — the existing helper in `audit.py`.
- **Admin auth:** All endpoints use `Depends(get_current_admin)` which verifies `is_system_admin or is_game_admin`.
- **Pagination:** All list endpoints accept `?page=1&page_size=50` and return `{"items": [...], "total": N, "page": P, "page_size": S}`.
- **Deletion blocking:** Delete endpoints check for child records and return 409 with `{"detail": "Cannot delete: 15 scenes reference this chapter", "child_counts": {"scenes": 15}}`.
- **Error responses:** 422 for validation, 404 for not found, 409 for constraint violations / deletion blocked. All include `{"detail": "Human-readable message"}`.
- **FK pickers:** Frontend uses searchable dropdowns backed by list endpoints. Cascading pickers (book → chapter → scene) filter child options based on parent selection.
- **Timestamps:** `created_at` and `updated_at` are system-managed via DB triggers. Never sent in requests.

### 1.4 Reusable Shared Components

These components are used across multiple editors and should be built first:

#### CascadingPicker
```typescript
interface CascadingPickerProps {
  level: 'book' | 'chapter' | 'scene' | 'beat';
  value: { bookId?: number; chapterId?: number; sceneId?: number; beatId?: number };
  onChange: (value: CascadingPickerValue) => void;
  optional?: boolean;  // whether leaf-level selection is required
}
```
Fetches books, then chapters filtered by book, then scenes filtered by chapter, then beats filtered by scene. Each dropdown only appears when its parent is selected.

#### DeleteConfirmDialog
```typescript
interface DeleteConfirmDialogProps {
  resourceType: string;       // "book", "chapter", etc.
  resourceName: string;       // display name
  childCounts: Record<string, number>;  // {"chapters": 5, "scenes": 42}
  isBlocked: boolean;         // true if childCounts > 0
  onConfirm: () => void;
  onCancel: () => void;
}
```
When `isBlocked`, the confirm button is disabled and the dialog shows "Cannot delete: X child records exist. Remove them first." When not blocked, shows "Are you sure? This cannot be undone." with a required "DELETE" text input confirmation.

#### JsonEditor
```typescript
interface JsonEditorProps {

  schema?: JsonSchema;         // optional schema for validation/rendering
```

```python
async def update_book(db, book_id, payload, admin_email, ip):
    book = db.get(Book, book_id)
    if not book: raise 404

    old_state = {f: getattr(book, f) for f in payload.model_fields_set}
    for field, value in payload.dict(exclude_unset=True).items():
        setattr(book, field, value)

    log_admin_action(db, admin_email, "book_updated", "book", book_id,

    db.commit()
    db.refresh(book)
    return book
```

### 2.4 Backend: `DELETE /api/admin/content/books/{id}`

**Service Logic:**
```python
async def delete_book(db, book_id, admin_email, ip):
    book = db.get(Book, book_id)
    if not book: raise 404

    chapter_count = db.exec(
        select(func.count()).where(Chapter.book_id == book_id)
    ).one()
    if chapter_count > 0:
        raise HTTPException(409, detail=f"Cannot delete: {chapter_count} chapters reference this book",
                           headers={"X-Child-Counts": json.dumps({"chapters": chapter_count})})

    log_admin_action(db, admin_email, "book_deleted", "book", book_id,
                     {"title": book.title, "book_number": book.book_number}, ip)
    db.delete(book)
    db.commit()
```

```python
async def create_scene(db, payload, admin_email, ip):
    # Validate chapter exists
    chapter = db.get(Chapter, payload.chapter_id)
    if not chapter: raise 404("Chapter not found")

    # Validate unique constraint
    existing = db.exec(
        select(Scene).where(Scene.chapter_id == payload.chapter_id,
                            Scene.scene_number == payload.scene_number)
    ).first()
    if existing: raise 409("Scene number already exists in this chapter")

    # Validate location if provided
    if payload.primary_location_id:
        loc = db.get(Location, payload.primary_location_id)
        if not loc: raise 404("Location not found")

    scene = Scene(**payload.dict(exclude={"gameplay_data", "boss_config"}))
    db.add(scene)
    db.flush()  # get scene.id

    # Auto-create scene_gameplay_data
    gd = SceneGameplayData(
        scene_id=scene.id,
        **payload.gameplay_data.dict() if payload.gameplay_data else {}
    )
    db.add(gd)

    log_admin_action(db, admin_email, "scene_created", "scene", scene.id,
                     {"title": scene.title, "chapter_id": scene.chapter_id}, ip)
    db.commit()
    return scene_with_details(db, scene.id)
```

### 4.3 Backend: `PATCH /api/admin/content/scenes/{id}`

Updates scene core fields AND `scene_gameplay_data` in a single call. The `gameplay_data` sub-object and `boss_config` are both optional — only provided fields are updated.

```json
    "timer_seconds": 120,
```

**Service Logic:**
```python
async def update_scene(db, scene_id, payload, admin_email, ip):
    scene = db.get(Scene, scene_id)
    if not scene: raise 404

    old_state = snapshot(scene)

    # Update core fields
    for field in ["title", "summary", "scene_number", "sort_order", "chapter_id",
                  "primary_location_id", "has_hard_break", "scene_type"]:
        if field in payload.model_fields_set:
            setattr(scene, field, getattr(payload, field))

    # Update boss_config if provided
    if "boss_config" in payload.model_fields_set:
        validate_boss_config(db, payload.boss_config)  # check entity_id exists, valid types
        scene.boss_config = payload.boss_config

    # Update gameplay data if provided
    if payload.gameplay_data is not None:
        gd = db.exec(select(SceneGameplayData).where(
            SceneGameplayData.scene_id == scene_id)).first()
        if not gd:
            gd = SceneGameplayData(scene_id=scene_id)
            db.add(gd)
        for field, value in payload.gameplay_data.dict(exclude_unset=True).items():
            setattr(gd, field, value)

    log_admin_action(db, admin_email, "scene_updated", "scene", scene_id,
                     {"old": old_state, "changes": payload.dict(exclude_unset=True)}, ip)
    db.commit()
    return scene_with_details(db, scene_id)


def validate_boss_config(db, config):
    """Validate boss_config structure."""
    if config is None:
        return
    if "entity_id" in config and config["entity_id"]:
        entity = db.get(Entity, config["entity_id"])
        if not entity:
            raise HTTPException(422, "boss_config.entity_id references non-existent entity")
    valid_types = {"click_burst", "target_zone", "whack_sequence"}
    for interrupt in config.get("interrupt_types", []):
        if interrupt.get("type") not in valid_types:
            raise HTTPException(422, f"Invalid interrupt type: {interrupt.get('type')}")
```

```python
async def set_entity_attack_types(db, entity_id, payload, admin_email, ip):
    entity = db.get(Entity, entity_id)
    if not entity: raise 404

    # Validate at most one primary
    primary_count = sum(1 for at in payload.attack_types if at.is_primary)
    if primary_count > 1:
        raise 422("At most one attack type can be marked as primary")

    # Delete existing assignments
    old_types = db.exec(
        select(EntityAttackType).where(EntityAttackType.entity_id == entity_id)
    ).all()
    old_snapshot = [{"attack_type_id": t.attack_type_id, "is_primary": t.is_primary} for t in old_types]
    for t in old_types:
        db.delete(t)

    # Create new assignments
    for at in payload.attack_types:
        db.add(EntityAttackType(
            entity_id=entity_id,
            attack_type_id=at.attack_type_id,
            is_primary=at.is_primary
        ))

    log_admin_action(db, admin_email, "entity_attack_types_set", "entity", entity_id,
                     {"old": old_snapshot, "new": [at.dict() for at in payload.attack_types]}, ip)
    db.commit()
```

```python
async def bulk_create_entity_scenes(db, payload, admin_email, ip):

                EntitySceneAppearance.scene_id == entry.scene_id
```

```python
async def upsert_wave_config(db, scene_id, payload, admin_email, ip):
    scene = db.get(Scene, scene_id)
    if not scene: raise 404

    # Validate entity_pool entity IDs exist
    for entry in payload.entity_pool:
        entity = db.get(Entity, entry["entity_id"])
        if not entity:
            raise 422(f"entity_pool contains non-existent entity_id: {entry['entity_id']}")

    existing = db.exec(
        select(SceneWaveConfig).where(SceneWaveConfig.scene_id == scene_id)
    ).first()

    if existing:
        old_state = snapshot(existing)
        for field, value in payload.dict().items():
            setattr(existing, field, value)
        action = "wave_config_updated"
    else:
        existing = SceneWaveConfig(scene_id=scene_id, **payload.dict())
        db.add(existing)
        old_state = None
        action = "wave_config_created"

    log_admin_action(db, admin_email, action, "scene", scene_id,
                     {"old": old_state, "new": payload.dict()}, ip)
    db.commit()
    return existing
```

```python
async def apply_wave_template(db, payload, admin_email, ip):
    if payload.scope == "chapter":
        scenes = db.exec(
            select(Scene).where(Scene.chapter_id == payload.scope_id)
            .order_by(Scene.sort_order)
        ).all()
    elif payload.scope == "book":
        chapter_ids = db.exec(
            select(Chapter.id).where(Chapter.book_id == payload.scope_id)
            .order_by(Chapter.sort_order)
        ).all()
        scenes = db.exec(
            select(Scene).where(Scene.chapter_id.in_([c.id for c in chapter_ids]))
            .order_by(Scene.chapter_id, Scene.sort_order)
        ).all()
    else:
        raise 422("scope must be 'chapter' or 'book'")

    created = 0
    updated = 0
    skipped = 0
    for i, scene in enumerate(scenes):
        existing = db.exec(
            select(SceneWaveConfig).where(SceneWaveConfig.scene_id == scene.id)
        ).first()

        if existing and not payload.overwrite_existing:
            skipped += 1
            continue

        config_data = {
            **payload.base_config,
            "scaling_factor": payload.scaling.scaling_factor_start + (i * payload.scaling.scaling_factor_increment),
            "hp_multiplier": payload.scaling.hp_multiplier_start + (i * payload.scaling.hp_multiplier_increment),
            "gold_multiplier": payload.scaling.gold_multiplier_start + (i * payload.scaling.gold_multiplier_increment),
        }

        if existing:
            for field, value in config_data.items():
                setattr(existing, field, value)
            updated += 1
        else:
            db.add(SceneWaveConfig(scene_id=scene.id, **config_data))
            created += 1

    log_admin_action(db, admin_email, "wave_config_template_applied", "wave_config", None,
                     {"scope": payload.scope, "scope_id": payload.scope_id,
                      "created": created, "updated": updated, "skipped": skipped}, ip)
    db.commit()
    return {"created": created, "updated": updated, "skipped": skipped, "total_scenes": len(scenes)}
```

### 10.3 Backend: `POST /api/admin/content/wave-configs/auto-populate/{scene_id}`

Auto-generates `entity_pool` from `entity_scene_appearances` for a scene.

```python
async def auto_populate_wave_config(db, scene_id, admin_email, ip):
    appearances = db.exec(
        select(EntitySceneAppearance).where(
            EntitySceneAppearance.scene_id == scene_id,
            EntitySceneAppearance.is_present == True
        )
    ).all()

    # Only include entities with gameplay data
    entity_pool = []
    for app in appearances:
        has_gameplay = db.exec(
            select(EntityGameplayData).where(EntityGameplayData.entity_id == app.entity_id)
        ).first()
        if has_gameplay:
            entity_pool.append({
                "entity_id": app.entity_id,
                "weight": 1.0,
                "min_wave": 1,
                "max_wave": None
            })

    config = db.exec(
        select(SceneWaveConfig).where(SceneWaveConfig.scene_id == scene_id)
    ).first()
    if config:
        config.entity_pool = entity_pool
    else:
        config = SceneWaveConfig(scene_id=scene_id, entity_pool=entity_pool)
        db.add(config)

    log_admin_action(db, admin_email, "wave_config_auto_populated", "scene", scene_id,
                     {"entity_count": len(entity_pool)}, ip)
    db.commit()
    return config
```
