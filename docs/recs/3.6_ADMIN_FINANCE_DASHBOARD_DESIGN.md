## 1. Backend Architecture

### 1.1 Module Structure

```
backend/
├── routes/
│   ├── admin_finance.py              # NEW: All 3.6 admin finance endpoints
│   ├── admin_payments.py             # EXISTING: Reconciliation, polling, webhooks
│   ├── admin_subscriptions.py        # EXISTING: Subscription lifecycle management
│   ├── admin_shop.py                 # EXISTING: Shop item/bundle CRUD
│   ├── admin_donations.py            # EXISTING: Donation listing/stats
│   └── admin_marketplace.py          # EXISTING: Marketplace moderation
├── services/
│   └── finance_analytics_service.py  # NEW: DB-driven aggregation queries
└── models/
    └── finance.py                    # NEW: AdminShardAdjustment model
```

### 1.2 Key Design Decisions

#### Single New Route File
All new 3.6 endpoints go into a single `admin_finance.py` route file with prefix `/api/admin/finance`. This keeps the new analytics/management endpoints separate from the existing domain-specific admin routes (which continue to serve their original purpose).

#### Analytics Are DB-Driven
All revenue, subscription, shop, marketplace, and donation analytics are computed server-side via SQL aggregation queries. No client-side calculation from raw data. This ensures accuracy and scalability.

#### Reuse Existing Admin Routes
The frontend UI calls **both** the new `admin_finance` endpoints (for analytics and new workflows) **and** the existing domain-specific admin endpoints (for CRUD operations). For example, shop item CRUD still goes through `admin_shop.py`, but shop analytics come from `admin_finance.py`.

#### Recharts for Charts
The existing `Dashboard.tsx` uses `recharts`. All finance charts use the same library for consistency.

#### CSV Export — Client-Side
CSV export is generated client-side from the currently loaded page data. No server-side CSV generation endpoint. The `CsvExportButton` component takes the current table data and column definitions, generates a CSV blob, and triggers a download.

```python
from datetime import datetime, timezone, timedelta
from sqlalchemy import text
from sqlmodel import Session


def get_revenue_overview(session: Session) -> dict:
    """Revenue summary cards: today/week/month revenue, active subs, shard circulation, marketplace volume."""
    now = datetime.now(timezone.utc)
    today_start = now.replace(hour=0, minute=0, second=0, microsecond=0)
    week_start = today_start - timedelta(days=today_start.weekday())  # Monday
    month_start = today_start.replace(day=1)

    # Revenue by period (from payment_orders)
    revenue_query = text("""
        SELECT
            COALESCE(SUM(CASE WHEN completed_at >= :today THEN price_cents ELSE 0 END), 0) AS today_cents,
            COALESCE(SUM(CASE WHEN completed_at >= :week THEN price_cents ELSE 0 END), 0) AS week_cents,
            COALESCE(SUM(CASE WHEN completed_at >= :month THEN price_cents ELSE 0 END), 0) AS month_cents
        FROM payment_orders
        WHERE status = 'completed'
    """)
    rev = session.execute(revenue_query, {
        "today": today_start, "week": week_start, "month": month_start,
    }).one()

    # Active subscribers
    active_subs = session.execute(text(
        "SELECT COUNT(*) FROM player_subscriptions WHERE status = 'active'"
    )).scalar() or 0

    # Shard circulation
    shard_circulation = session.execute(text(
        "SELECT COALESCE(SUM(shard_balance), 0) FROM player_meta_progression"
    )).scalar() or 0

    # Marketplace volume (7d)
    marketplace_vol = session.execute(text("""
        SELECT COALESCE(SUM(price_shards), 0) FROM marketplace_trades
        WHERE traded_at >= :since
    """), {"since": now - timedelta(days=7)}).scalar() or 0

    return {
        "today_revenue_cents": rev.today_cents,
        "week_revenue_cents": rev.week_cents,
        "month_revenue_cents": rev.month_cents,
        "active_subscribers": active_subs,
        "shard_circulation": shard_circulation,
        "marketplace_volume_7d": marketplace_vol,
    }
```

### 2.2 Revenue Chart (Daily Breakdown)

```python
def get_revenue_chart(days: int, session: Session) -> list[dict]:
    """Daily revenue breakdown by source for the last N days.

    Note: payment_orders has NO order_type column. We derive source by
    checking whether a donation row references the order (donation),
    or a subscription stipend_log references it (subscription), else purchase.
    """
    query = text("""
        SELECT
            d.day::date AS day,
            COALESCE(SUM(
                CASE WHEN don.id IS NOT NULL THEN po.price_cents ELSE 0 END
            ), 0) AS donations_cents,
            COALESCE(SUM(
                CASE WHEN don.id IS NULL AND ssl.id IS NOT NULL THEN po.price_cents ELSE 0 END
            ), 0) AS subscriptions_cents,
            COALESCE(SUM(
                CASE WHEN don.id IS NULL AND ssl.id IS NULL THEN po.price_cents ELSE 0 END
            ), 0) AS purchases_cents
        FROM generate_series(
            CURRENT_DATE - :days * INTERVAL '1 day',
            CURRENT_DATE,
            '1 day'
        ) AS d(day)
        LEFT JOIN payment_orders po
            ON po.completed_at::date = d.day::date
            AND po.status = 'completed'
        LEFT JOIN donations don ON don.payment_order_id = po.id
        LEFT JOIN subscription_stipend_log ssl ON ssl.player_id = po.player_id
            AND ssl.created_at::date = po.completed_at::date
        GROUP BY d.day
        ORDER BY d.day
    """)
    rows = session.execute(query, {"days": days}).all()
    return [
        {
            "date": str(row.day),
            "purchases": row.purchases_cents / 100,
            "subscriptions": row.subscriptions_cents / 100,
            "donations": row.donations_cents / 100,
        }
        for row in rows
    ]
```

### 2.3 Shard Economy Metrics

```python
def get_shard_economy(session: Session) -> dict:
    """Shard economy health: circulation, purchased, spent, burned, net flow, spend rate."""
    query = text("""
        SELECT
            COALESCE(SUM(CASE WHEN amount > 0 AND source_type = 'purchase' THEN amount ELSE 0 END), 0) AS total_purchased,
            COALESCE(SUM(CASE WHEN amount < 0 THEN ABS(amount) ELSE 0 END), 0) AS total_spent,
            COALESCE(SUM(CASE WHEN source_type = 'admin_grant' THEN amount ELSE 0 END), 0) AS total_admin_granted,
            COALESCE(SUM(CASE WHEN source_type = 'admin_deduct' THEN ABS(amount) ELSE 0 END), 0) AS total_admin_debited,
            COALESCE(SUM(CASE WHEN source_type IN ('achievement', 'subscription_stipend', 'donation') AND amount > 0 THEN amount ELSE 0 END), 0) AS total_earned
        FROM shard_transactions
    """)
    row = session.execute(query).one()

    # Tax burned from marketplace
    tax_burned = session.execute(text(
        "SELECT COALESCE(SUM(tax_shards), 0) FROM marketplace_trades"
    )).scalar() or 0

    # Shard circulation
    circulation = session.execute(text(
        "SELECT COALESCE(SUM(shard_balance), 0) FROM player_meta_progression"
    )).scalar() or 0

    # 30d net flow
    net_flow_30d = session.execute(text("""
        SELECT COALESCE(SUM(amount), 0) FROM shard_transactions
        WHERE created_at >= NOW() - INTERVAL '30 days'
    """)).scalar() or 0

    total_purchased = row.total_purchased
    total_spent = row.total_spent
    spend_rate = (total_spent / total_purchased * 100) if total_purchased > 0 else 0

    return {
        "circulation": circulation,
        "total_purchased": total_purchased,
        "total_spent": total_spent,
        "total_earned": row.total_earned,
        "total_admin_granted": row.total_admin_granted,
        "total_admin_debited": row.total_admin_debited,
        "tax_burned": tax_burned,
        "net_flow_30d": net_flow_30d,
        "spend_rate_pct": round(spend_rate, 1),
    }
```

```python
def get_shard_flow_chart(days: int, session: Session) -> list[dict]:
    """Daily shard inflows vs outflows for the last N days."""
    query = text("""
        SELECT
            d.day::date AS day,
            COALESCE(SUM(CASE WHEN st.amount > 0 THEN st.amount ELSE 0 END), 0) AS inflows,
            COALESCE(SUM(CASE WHEN st.amount < 0 THEN ABS(st.amount) ELSE 0 END), 0) AS outflows
        FROM generate_series(
            CURRENT_DATE - :days * INTERVAL '1 day',
            CURRENT_DATE,
            '1 day'
        ) AS d(day)
        LEFT JOIN shard_transactions st
            ON st.created_at::date = d.day::date
        GROUP BY d.day
        ORDER BY d.day
    """)
    rows = session.execute(query, {"days": days}).all()
    return [{"date": str(row.day), "inflows": row.inflows, "outflows": row.outflows} for row in rows]
```

### 2.5 Subscription Metrics

```python
def get_subscription_metrics(session: Session) -> dict:
    """Subscription KPIs: active, plan split, churn (voluntary + involuntary), conversion, MRR.

    Model notes (from PlayerSubscription):
    - plan_key: 'monthly' | 'annual' (not plan_type)
    - status: 'active', 'canceling', 'past_due', 'expired' (no 'cancelled')
    - cancel_at_period_end: True = voluntary cancellation pending end-of-period
    - continuous_streak: int (not continuous_streak_months)
    - cumulative_subscription_months: tracked on players table, not subscription
    - No cancelled_at field — voluntary churn = status='canceling' or
      (status='expired' AND cancel_at_period_end=True)
    - Involuntary churn = status='expired' AND cancel_at_period_end=False
      (payment failure led to expiry without player cancellation)
    """
    now = datetime.now(timezone.utc)
    thirty_days_ago = now - timedelta(days=30)

    stats = session.execute(text("""
        SELECT
            COUNT(*) FILTER (WHERE status = 'active') AS active_count,
            COUNT(*) FILTER (WHERE status = 'active' AND plan_key = 'monthly') AS monthly_count,
            COUNT(*) FILTER (WHERE status = 'active' AND plan_key = 'annual') AS annual_count,
            COUNT(*) FILTER (WHERE status = 'past_due') AS past_due_count,
            COUNT(*) FILTER (WHERE status IN ('canceling', 'expired')
                AND cancel_at_period_end = TRUE
                AND updated_at >= :since) AS voluntary_churn_30d,
            COUNT(*) FILTER (WHERE status = 'expired'
                AND cancel_at_period_end = FALSE
                AND updated_at >= :since) AS involuntary_churn_30d,
            COUNT(*) AS total_ever
        FROM player_subscriptions
    """), {"since": thirty_days_ago}).one()

    # Active at start of period (approximation: current active + churned in period)
    active_start = stats.active_count + stats.voluntary_churn_30d + stats.involuntary_churn_30d
    vol_churn_rate = (stats.voluntary_churn_30d / active_start * 100) if active_start > 0 else 0
    invol_churn_rate = (stats.involuntary_churn_30d / active_start * 100) if active_start > 0 else 0

    total_players = session.execute(text("SELECT COUNT(*) FROM players")).scalar() or 1
    conversion_rate = (stats.total_ever / total_players * 100)

    # MRR: monthly × $1.99 + annual × ($19.90 / 12)
    mrr = (stats.monthly_count * 199 + stats.annual_count * 1658) / 100  # in dollars

    return {
        "active_count": stats.active_count,
        "monthly_count": stats.monthly_count,
        "annual_count": stats.annual_count,
        "past_due_count": stats.past_due_count,
        "voluntary_churn_30d": stats.voluntary_churn_30d,
        "involuntary_churn_30d": stats.involuntary_churn_30d,
        "total_churn_30d": stats.voluntary_churn_30d + stats.involuntary_churn_30d,
        "voluntary_churn_rate_pct": round(vol_churn_rate, 1),
        "involuntary_churn_rate_pct": round(invol_churn_rate, 1),
        "total_churn_rate_pct": round(vol_churn_rate + invol_churn_rate, 1),
        "conversion_rate_pct": round(conversion_rate, 1),
        "mrr_dollars": round(mrr, 2),
    }
```

```python
def get_shop_analytics(session: Session) -> dict:
    """Shop purchase stats: items sold, revenue by category, top items, active boosters."""
    # Total items sold
    total_sold = session.execute(text(
        "SELECT COUNT(*) FROM player_shop_items"
    )).scalar() or 0

    # Revenue by category
    category_revenue = session.execute(text("""
        SELECT si.category, COUNT(*) AS sold_count, SUM(si.price_shards) AS total_shards
        FROM player_shop_items psi
        JOIN shop_items si ON psi.shop_item_id = si.id
        GROUP BY si.category
        ORDER BY total_shards DESC
    """)).all()

    # Top 10 selling items
    top_items = session.execute(text("""
        SELECT si.name, si.category, si.price_shards, COUNT(*) AS sold_count
        FROM player_shop_items psi
        JOIN shop_items si ON psi.shop_item_id = si.id
        GROUP BY si.id, si.name, si.category, si.price_shards
        ORDER BY sold_count DESC
        LIMIT 10
    """)).all()

    # Active boosters
    active_boosters = session.execute(text("""
        SELECT COUNT(*) FROM player_active_boosters
        WHERE status = 'active'
    """)).scalar() or 0

    return {
        "total_items_sold": total_sold,
        "category_revenue": [
            {"category": r.category, "sold_count": r.sold_count, "total_shards": r.total_shards}
            for r in category_revenue
        ],
        "top_items": [
            {"name": r.name, "category": r.category, "price_shards": r.price_shards, "sold_count": r.sold_count}
            for r in top_items
        ],
        "active_boosters": active_boosters,
    }
```

```python
def get_marketplace_analytics(session: Session) -> dict:
    """Enhanced marketplace stats: volume trends, unique traders, listing counts."""
    now = datetime.now(timezone.utc)

    stats = session.execute(text("""
        SELECT
            COUNT(*) AS total_trades,
            COALESCE(SUM(price_shards), 0) AS total_volume,
            COALESCE(AVG(price_shards), 0) AS avg_price,
            COALESCE(SUM(tax_shards), 0) AS total_tax_burned,
            COUNT(*) FILTER (WHERE traded_at >= :seven_d) AS trades_7d,
            COALESCE(SUM(price_shards) FILTER (WHERE traded_at >= :seven_d), 0) AS volume_7d,
            COUNT(*) FILTER (WHERE traded_at >= :thirty_d) AS trades_30d,
            COALESCE(SUM(price_shards) FILTER (WHERE traded_at >= :thirty_d), 0) AS volume_30d
        FROM marketplace_trades
    """), {"seven_d": now - timedelta(days=7), "thirty_d": now - timedelta(days=30)}).one()

    # Active listings
    active_listings = session.execute(text(
        "SELECT COUNT(*) FROM marketplace_listings WHERE status = 'active' AND expires_at > NOW()"
    )).scalar() or 0

    # Unique traders (30d)
    unique_traders = session.execute(text("""
        SELECT COUNT(DISTINCT player_id) FROM (
            SELECT buyer_id AS player_id FROM marketplace_trades WHERE traded_at >= :since
            UNION
            SELECT seller_id AS player_id FROM marketplace_trades WHERE traded_at >= :since
        ) sub
    """), {"since": now - timedelta(days=30)}).scalar() or 0

    # Listings created (30d)
    listings_created_30d = session.execute(text("""
        SELECT COUNT(*) FROM marketplace_listings WHERE listed_at >= :since
    """), {"since": now - timedelta(days=30)}).scalar() or 0

    return {
        "active_listings": active_listings,
        "total_trades": stats.total_trades,
        "total_volume": stats.total_volume,
        "avg_price": round(float(stats.avg_price), 1),
        "total_tax_burned": stats.total_tax_burned,
        "trades_7d": stats.trades_7d,
        "volume_7d": stats.volume_7d,
        "trades_30d": stats.trades_30d,
        "volume_30d": stats.volume_30d,
        "unique_traders_30d": unique_traders,
        "listings_created_30d": listings_created_30d,
    }
```

```python
def get_donation_analytics(session: Session) -> dict:
    """Enhanced donation stats: repeat rate, tier distribution, top donors."""
    # Summary
    summary = session.execute(text("""
        SELECT
            COUNT(*) AS total_donations,
            COUNT(DISTINCT player_id) AS total_donors,
            COALESCE(SUM(amount_cents), 0) AS total_cents,
            COALESCE(AVG(amount_cents), 0) AS avg_cents,
            COALESCE(SUM(amount_cents) FILTER (WHERE created_at >= NOW() - INTERVAL '30 days'), 0) AS total_30d_cents
        FROM donations
    """)).one()

    # Repeat donor rate
    repeat_donors = session.execute(text("""
        SELECT COUNT(*) FROM (
            SELECT player_id FROM donations GROUP BY player_id HAVING COUNT(*) > 1
        ) sub
    """)).scalar() or 0
    repeat_rate = (repeat_donors / summary.total_donors * 100) if summary.total_donors > 0 else 0

    # Patron tier distribution
    tier_dist = session.execute(text("""
        SELECT patron_tier, COUNT(*) AS count
        FROM players
        WHERE patron_tier IS NOT NULL AND patron_tier != ''
        GROUP BY patron_tier
        ORDER BY count DESC
    """)).all()

    # Top 10 donors
    top_donors = session.execute(text("""
        SELECT p.display_name, p.id AS player_id, SUM(d.amount_cents) AS total_cents, COUNT(*) AS donation_count
        FROM donations d
        JOIN players p ON d.player_id = p.id
        GROUP BY p.id, p.display_name
        ORDER BY total_cents DESC
        LIMIT 10
    """)).all()

    return {
        "total_donations": summary.total_donations,
        "total_donors": summary.total_donors,
        "total_cents": summary.total_cents,
        "avg_cents": round(float(summary.avg_cents)),
        "total_30d_cents": summary.total_30d_cents,
        "repeat_donor_rate_pct": round(repeat_rate, 1),
        "tier_distribution": [
            {"tier": r.patron_tier, "count": r.count} for r in tier_dist
        ],
        "top_donors": [
            {"display_name": r.display_name, "player_id": r.player_id,
             "total_cents": r.total_cents, "donation_count": r.donation_count}
            for r in top_donors
        ],
    }
```

### 2.9 Marketplace Anomaly Detection

```python
def detect_marketplace_anomalies(session: Session) -> dict:
    """Detect price anomalies, rapid relists, and wash trade suspicions."""
    # Load thresholds from game_configs
    price_mult = _get_config_float(session, "anomaly_price_multiplier_threshold", 10.0)
    relist_count = _get_config_int(session, "anomaly_rapid_relist_count", 5)
    relist_window = _get_config_int(session, "anomaly_rapid_relist_window_minutes", 60)
    wash_count = _get_config_int(session, "anomaly_wash_trade_count", 3)
    wash_window = _get_config_int(session, "anomaly_wash_trade_window_hours", 24)

    # 1. High-price anomalies: listings > Nx average for same item_name + item_rarity
    price_anomalies = session.execute(text("""
        SELECT ml.id, ml.seller_id, ml.item_name, ml.item_rarity, ml.price_shards,
               avgs.avg_price, ml.price_shards::float / NULLIF(avgs.avg_price, 0) AS multiplier
        FROM marketplace_listings ml
        JOIN (
            SELECT item_name, item_rarity, AVG(price_shards) AS avg_price
            FROM marketplace_listings
            WHERE status IN ('active', 'sold') AND listed_at >= NOW() - INTERVAL '30 days'
            GROUP BY item_name, item_rarity
            HAVING COUNT(*) >= 3
        ) avgs ON ml.item_name = avgs.item_name AND ml.item_rarity = avgs.item_rarity
        WHERE ml.status = 'active'
          AND ml.price_shards > avgs.avg_price * :threshold
        ORDER BY multiplier DESC
        LIMIT 50
    """), {"threshold": price_mult}).all()

    # 2. Rapid relists: same seller, same item_type, > N listings in window
    rapid_relists = session.execute(text("""
        SELECT seller_id, item_name, COUNT(*) AS listing_count,
               MIN(listed_at) AS first_listed, MAX(listed_at) AS last_listed
        FROM marketplace_listings
        WHERE listed_at >= NOW() - :window * INTERVAL '1 minute'
        GROUP BY seller_id, item_name
        HAVING COUNT(*) >= :count
        ORDER BY listing_count DESC
    """), {"window": relist_window, "count": relist_count}).all()

    # 3. Wash trade suspicion: same two players trading back and forth > N times in window
    wash_trades = session.execute(text("""
        SELECT buyer_id, seller_id, COUNT(*) AS trade_count
        FROM marketplace_trades
        WHERE traded_at >= NOW() - :window * INTERVAL '1 hour'
        GROUP BY LEAST(buyer_id, seller_id), GREATEST(buyer_id, seller_id), buyer_id, seller_id
        HAVING COUNT(*) >= :count
        ORDER BY trade_count DESC
```

```python
    body: ShardAdjustRequest,
    request: Request,
    token: dict = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """Manual shard grant or debit for support cases."""
    if not body.reason.strip():
        raise HTTPException(status_code=400, detail="Reason is required.")
    if body.amount <= 0:
        raise HTTPException(status_code=400, detail="Amount must be positive.")
    if body.adjust_type not in ('grant', 'debit'):
        raise HTTPException(status_code=400, detail="adjust_type must be 'grant' or 'debit'.")

    # Verify player exists
    player = session.execute(
        text("SELECT id, display_name, shard_balance FROM player_meta_progression pm JOIN players p ON pm.player_id = p.id WHERE p.id = :pid"),
        {"pid": body.player_id}
    ).first()
    if not player:
        raise HTTPException(status_code=404, detail="Player not found.")

    admin_email = token.get("email", "unknown")

    if body.adjust_type == 'grant':
        credit_shards_direct(
            player_id=body.player_id,
            amount=body.amount,
            source_type="admin_grant",
            description=f"Admin grant by {admin_email}: {body.reason}",
            session=session,
        )
    else:
        debit_shards(
            player_id=body.player_id,
            amount=body.amount,
            source_type="admin_deduct",
            source_id=None,
            description=f"Admin debit by {admin_email}: {body.reason}",
            session=session,
        )

    session.commit()

    # Fetch new balance and the shard_transactions record for the audit table
    new_balance = session.execute(
        text("SELECT shard_balance FROM player_meta_progression WHERE player_id = :pid"),
        {"pid": body.player_id}
    ).scalar()
    latest_txn = session.execute(
        text("SELECT id FROM shard_transactions WHERE player_id = :pid ORDER BY id DESC LIMIT 1"),
        {"pid": body.player_id}
    ).scalar()

    # Write to admin_shard_adjustments audit table
    session.execute(text("""
        INSERT INTO admin_shard_adjustments
            (player_id, admin_email, adjust_type, amount, reason, balance_before, balance_after, shard_txn_id)
        VALUES (:pid, :email, :atype, :amount, :reason, :before, :after, :txn_id)
    """), {
        "pid": body.player_id,
        "email": admin_email,
        "atype": body.adjust_type,
        "amount": body.amount,
        "reason": body.reason,
        "before": player.shard_balance,
        "after": new_balance,
        "txn_id": latest_txn,
    })
    session.commit()

    write_audit_log(
        session=session,
        admin_email=admin_email,
        action=f"shard_{body.adjust_type}",
        target_type="player",
        target_id=str(body.player_id),
        details={"amount": body.amount, "type": body.adjust_type, "reason": body.reason},
        ip=get_client_ip(request),
    )

    return {"status": "ok", "player_id": body.player_id, "adjusted": body.amount, "type": body.adjust_type}

@router.post("/initiate-refund")
async def initiate_refund(
    body: InitiateRefundRequest,
    request: Request,
    token: dict = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """Create a Stripe refund and auto-debit proportional shards."""
    from models.payments import PaymentOrder

    order = session.get(PaymentOrder, body.order_id)
    if not order:
        raise HTTPException(status_code=404, detail="Payment order not found.")
    if order.status not in ('completed',):
        raise HTTPException(status_code=400, detail=f"Cannot refund order with status '{order.status}'.")
    if not order.stripe_charge_id:
        raise HTTPException(status_code=400, detail="Order has no Stripe charge ID.")
    if not body.reason.strip():
        raise HTTPException(status_code=400, detail="Reason is required.")

    # Determine refund amount
    refund_cents = body.amount_cents if body.amount_cents else order.price_cents
    already_refunded_cents = int((order.shards_refunded / max(order.shards_credited, 1)) * order.price_cents)
    max_refundable = order.price_cents - already_refunded_cents

    if refund_cents > max_refundable:
        raise HTTPException(status_code=400, detail=f"Max refundable is ${max_refundable / 100:.2f}.")

    # Call Stripe API
    try:
        stripe_refund = stripe.Refund.create(
            charge=order.stripe_charge_id,
            amount=refund_cents,
            reason="requested_by_customer",
            metadata={"admin_email": token.get("email"), "reason": body.reason},
        )
    except stripe.error.StripeError as e:
        raise HTTPException(status_code=502, detail=f"Stripe error: {str(e)}")

    # Calculate proportional shard debit
    refund_shards = calculate_refund_shards(order.shards_credited, order.price_cents, refund_cents + already_refunded_cents)
    new_shard_debit = refund_shards - order.shards_refunded

    if new_shard_debit > 0:
        debit_shards(
            player_id=order.player_id,
            amount=new_shard_debit,
            source_type="refund",
            source_id=order.id,
            description=f"Admin-initiated refund for order #{order.id}: {body.reason}",
            session=session,
        )
        order.shards_refunded += new_shard_debit

    if not order.refunded_at:
        order.refunded_at = datetime.now(timezone.utc)
    if order.shards_refunded >= order.shards_credited:
        order.status = "refunded"

    session.add(order)
    session.commit()

    write_audit_log(
        session=session,
        admin_email=token.get("email", "unknown"),
        action="initiate_refund",
        target_type="payment_order",
        target_id=str(order.id),
        details={
            "refund_cents": refund_cents,
            "shards_debited": new_shard_debit,
            "stripe_refund_id": stripe_refund.id,
            "reason": body.reason,
        },
        ip=get_client_ip(request),
    )

    return {
        "status": "ok",
        "stripe_refund_id": stripe_refund.id,
        "refund_cents": refund_cents,
        "shards_debited": new_shard_debit,
    }
```

```tsx
        const val = row[c.key];
        const str = val === null || val === undefined ? '' : String(val);
        // Escape commas and quotes
        return str.includes(',') || str.includes('"')
          ? `"${str.replace(/"/g, '""')}"`
          : str;
      }).join(',')
    );
    const csv = [header, ...rows].join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${filename}_${new Date().toISOString().slice(0, 10)}.csv`;
    a.click();
    URL.revokeObjectURL(url);
  };

  return (
    <button className="btn-csv-export" onClick={handleExport} disabled={data.length === 0}>
      ⬇ Export CSV
    </button>
  );
}
```
