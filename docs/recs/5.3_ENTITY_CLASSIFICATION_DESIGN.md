### 1.2 Shared Patterns

- **Concurrency model:** Last-write-wins. No optimistic locking.
- **Audit logging:** All mutations call `write_audit_log(session, admin_email, action, target_type, target_id, details, ip)` — the existing helper in `audit.py`.
- **Admin auth:** All endpoints use `Depends(get_current_admin)` which verifies `is_system_admin or is_game_admin`.
- **Pagination:** All list endpoints accept `?page=1&page_size=50` and return `{"items": [...], "total": N, "page": P, "page_size": S}`.
- **Deletion blocking:** Delete endpoints check for child/FK records and return 409 with `{"detail": "Cannot delete: N entities reference this type", "child_counts": {"entities": N}}`.
- **Error responses:** 422 for validation, 404 for not found, 409 for constraint violations / deletion blocked. All include `{"detail": "Human-readable message"}`.

### 1.4 Data Migration Strategy

The migration (§Schema) performs a **clean break**:
1. Create new lookup tables (`entity_types`, `entity_families`, `visual_behaviors`).
2. Seed from existing data.
3. Add FK columns (`entity_type_id`, `entity_family_id`) to `entities`.
4. Populate FKs from existing string values.
5. Drop old string columns (`entity_type`, `entity_family`).
6. Add new columns to `attack_types` (`visual_behavior_id`, `stat_multipliers`).

    types = session.exec(

```python
def delete_entity_type(session: Session, type_id: int) -> None:
    entity_type = session.get(EntityType, type_id)
    if not entity_type:
        raise HTTPException(404, detail="Entity type not found")

    entity_count = session.exec(
        select(func.count()).select_from(Entity)
        .where(Entity.entity_type_id == type_id)
    ).one()
    if entity_count > 0:
        raise HTTPException(
            409,
            detail=f"Cannot delete: {entity_count} entities reference this type"
        )

    session.delete(entity_type)
    session.commit()
```

def _get_transitive_entity_count(session: Session, behavior_id: int) -> int:
    """Count entities that inherit this visual behavior via their primary attack type."""
    # Get attack type IDs mapped to this behavior
    at_ids = session.exec(
        select(AttackType.id).where(AttackType.visual_behavior_id == behavior_id)
    ).all()
    if not at_ids:
        return 0

    # Count entities with a primary attack type in that set
    return session.exec(
        select(func.count()).select_from(EntityAttackType)
        .where(EntityAttackType.is_primary == True)
        .where(col(EntityAttackType.attack_type_id).in_(at_ids))
    ).one()
```

### 4.3 Animation Config Schema

The `animation_config` JSONB field has the following recognized keys (all optional with defaults):

| Key | Type | Default | Description |
|-----|------|---------|-------------|
| `y_offset` | integer | 0 | Vertical pixel offset from ground level (negative = higher) |
| `movement_pattern` | string | "walk" | Movement behavior: "walk", "hover", "teleport", "stationary", "burrow" |
| `attack_animation` | string | "melee_swing" | Attack animation key: "melee_swing", "projectile", "spell_cast", "dive_attack", "area_blast" |
| `attack_vfx` | string | null | Optional VFX overlay key during attack |
| `idle_animation` | string | "idle_bounce" | Idle state animation: "idle_bounce", "float", "pulse", "shimmer" |
| `speed_modifier` | float | 1.0 | Movement speed multiplier relative to base |

**Note:** These keys define the data contract. Actual rendering is deferred (§9 of requirements). New keys can be added without migration — the field is JSONB.

**Supported actions:**

| Action | Payload | Description |
|--------|---------|-------------|
| `assign_type` | `{"entity_type_id": N}` | Set entity type on all selected entities |
| `assign_family` | `{"entity_family_id": N}` | Set family on all selected entities |
| `add_attack_types` | `{"attack_type_ids": [N, ...]}` | Add attack types (preserves existing) |
| `replace_attack_types` | `{"attack_type_ids": [N, ...], "primary_id": N}` | Replace all attack types, set primary |
| `set_primary` | `{"attack_type_id": N}` | Set primary attack type (must already be assigned) |
| `create_family` | `{"family_name": "...", "family_display_name": "..."}` | Create new family + assign all selected |

**Response:**

{
  "success_count": 5,
  "skip_count": 0,
  "error_count": 0,
  "errors": []
}
```

**Service Logic:**
```python
def bulk_assign(session: Session, entity_ids: list[int], action: str, payload: dict) -> dict:
    success = 0
    skipped = 0
    errors = []

    entities = session.exec(
        select(Entity).where(col(Entity.id).in_(entity_ids))
    ).all()
    entity_map = {e.id: e for e in entities}

    for eid in entity_ids:
        entity = entity_map.get(eid)
        if not entity:
            errors.append({"entity_id": eid, "error": "Entity not found"})
            continue

        try:
            if action == "assign_type":
                entity.entity_type_id = payload["entity_type_id"]
                entity.updated_at = _now()
                session.add(entity)
                success += 1
            elif action == "assign_family":
                entity.entity_family_id = payload["entity_family_id"]
                entity.updated_at = _now()
                session.add(entity)
                success += 1
            elif action == "add_attack_types":
                _bulk_add_attack_types(session, eid, payload["attack_type_ids"])
                success += 1
            elif action == "replace_attack_types":
                _bulk_replace_attack_types(session, eid, payload["attack_type_ids"], payload.get("primary_id"))
                success += 1
            elif action == "set_primary":
                _bulk_set_primary(session, eid, payload["attack_type_id"])
                success += 1
        except Exception as e:
            errors.append({"entity_id": eid, "error": str(e)})

    session.commit()
    return {"success_count": success, "skip_count": skipped, "error_count": len(errors), "errors": errors}
```

**Service Logic:**
```python
def apply_template(session: Session, entity_ids: list[int], preview_only: bool) -> dict:
    previews = []

    for eid in entity_ids:
        entity = session.get(Entity, eid)
        if not entity:
            continue

        family = session.get(EntityFamily, entity.entity_family_id) if entity.entity_family_id else None
        primary_eat = session.exec(
            select(EntityAttackType)
            .where(EntityAttackType.entity_id == eid)
            .where(EntityAttackType.is_primary == True)
        ).first()
        attack_type = session.get(AttackType, primary_eat.attack_type_id) if primary_eat else None

        base = family.base_stat_template if family and family.base_stat_template else None
        multipliers = attack_type.stat_multipliers if attack_type and attack_type.stat_multipliers else {}

        warnings = []
        can_apply = True
        proposed = None

        if not base:
            warnings.append("No family assigned — cannot compute base template" if not family
                          else "Family has no base_stat_template defined")
            can_apply = False
        else:
            proposed = {}
            for key in ["strength", "agility", "intelligence"]:
                proposed[key] = round(base.get(key, 0) * multipliers.get(key, 1.0))

        proposed_hp = round(base.get("base_hp", 0) * multipliers.get("hp_multiplier", 1.0)) if base else None
        proposed_gold = round(base.get("base_gold", 0) * multipliers.get("gold_multiplier", 1.0)) if base else None

        gp = session.exec(
            select(EntityGameplayData).where(EntityGameplayData.entity_id == eid)
        ).first()

        preview = {
            "entity_id": eid,
            "entity_name": entity.canonical_name,
            "family_name": family.name if family else None,
            "primary_attack_type": attack_type.name if attack_type else None,
            "current_stat_block": gp.stat_block if gp else None,
            "proposed_stat_block": proposed,
            "current_base_hp": gp.base_hp if gp else 0,
            "proposed_base_hp": proposed_hp,
            "current_base_gold": gp.base_gold if gp else 0,
            "proposed_base_gold": proposed_gold,
            "can_apply": can_apply,
            "warnings": warnings,
        }
        previews.append(preview)

        if not preview_only and can_apply:
            now = _now()
            if not gp:
                gp = EntityGameplayData(entity_id=eid, created_at=now, updated_at=now)
            gp.stat_block = proposed
            gp.base_hp = proposed_hp
            gp.base_gold = proposed_gold
            gp.updated_at = now
            session.add(gp)

    if not preview_only:
        session.commit()

    applicable = sum(1 for p in previews if p["can_apply"])
    return {
        "previews": previews,
        "applicable_count": applicable,
        "skipped_count": len(previews) - applicable,
    }
```
