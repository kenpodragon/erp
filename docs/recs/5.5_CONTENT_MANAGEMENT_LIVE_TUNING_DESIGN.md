### 1.2 Key Principle: No New Backend Endpoints

5.5 adds **no new backend endpoints**. All specialized panels read from and write to the existing GameConfigs CRUD endpoints:

- `GET /api/admin/game-configs` — Fetch all configs (existing)
- `PATCH /api/admin/game-configs/{key}` — Update a single config (existing)

The Skill Balance tab also reads from:
- `GET /api/admin/content/skills` — Fetch all skills (existing, from 5.2)

All computation (DPS calculations, distribution previews, XP curve rendering) happens **client-side**.

### 1.3 Shared Patterns

- **Admin auth:** Page already gated behind admin role check.
- **Optimistic saves:** Individual key updates via existing PATCH endpoint. Success toast per save.
- **Reactive preview:** Specialized panels compute previews from local state (pre-save). "Save" writes to server; "Reset" restores server values.
- **Type inference:** `inferInputType(value_json)` utility function used by both the All Configs view and specialized panels.

### 2.2 Frontend: `TypeAwareInput.tsx`

```typescript
// tuning-utils.ts
type InferredType = 'slider_01' | 'integer' | 'float' | 'string' | 'boolean' | 'json_array' | 'json_object';

function inferInputType(value_json: any): InferredType {
  if (typeof value_json === 'boolean') return 'boolean';
  if (typeof value_json === 'string') return 'string';
  if (Array.isArray(value_json)) return 'json_array';
  if (typeof value_json === 'object' && value_json !== null) return 'json_object';
  if (typeof value_json === 'number') {
    if (Number.isInteger(value_json) && value_json >= 0) return 'integer';
    if (value_json >= 0 && value_json <= 1) return 'slider_01';
    return 'float';
  }
  return 'string'; // fallback
}
```

**Component behavior:**
- `slider_01`: Horizontal range slider (0.0–1.0, step 0.01) + number input showing value. Both linked.
- `integer`: Number input with step=1, min=0.
- `float`: Number input with step=0.01.
- `string`: Text input.
- `boolean`: Toggle switch.
- `json_array` / `json_object`: Collapsible JSON editor with syntax highlighting (uses existing pattern from GameConfigs).

### 2.3 Frontend: `ConfigSearchBar.tsx`

```typescript
function searchConfigs(configs: GameConfig[], query: string): GameConfig[] {
  const q = query.toLowerCase();
  return configs.filter(c =>
    c.key.toLowerCase().includes(q) ||
    (c.description || '').toLowerCase().includes(q) ||
    (c.category || '').toLowerCase().includes(q)
  );
}
```

**Behavior:**
- Typing filters across all categories simultaneously.
- Matching categories get a highlight badge on their tab.
- Results show with category context: "combat > hp_scaling_factor".
- Click result to jump to that category tab and scroll to the key.

### 3.3 Frontend: `DropPreview.tsx`

```typescript
function computeDropPreview(configs: Record<string, any>): DropPreviewData {
  const sceneChance = configs['artifact_gen_drop_chance_scene'] ?? 0.05;
  const bossChance = configs['artifact_gen_drop_chance_boss'] ?? 0.10;
  const masteryChance = configs['artifact_gen_drop_chance_mastery'] ?? 0.15;
  const bossRate = 0.2; // roughly 1 in 5 scenes has a boss

  const per100Scenes = {
    from_scenes: Math.round(100 * sceneChance * 10) / 10,
    from_bosses: Math.round(100 * bossRate * bossChance * 10) / 10,
    from_mastery: Math.round(100 * 0.05 * masteryChance * 10) / 10, // ~5% of scenes are mastery
  };

  return { per100Scenes };
}
```

```typescript
interface SkillBalanceRow {
  id: number;
  name: string;
  display_name: string;
  category: string;
  class_name: string | null;
  base_cooldown_seconds: number;
  base_cost_gold: number;
  cost_scaling_factor: number;
  cost_at_5: number;     // computed
  cost_at_10: number;    // computed
  cost_at_25: number;    // computed
  eff_cd_at_10: number;  // computed
  eff_cd_at_25: number;  // computed
}

// tuning-utils.ts
function computeSkillBalance(
  skill: Skill,
  coefficients: Record<string, number>
): SkillBalanceRow {
  const cdReduction = coefficients['cd_reduction_per_level'] ?? 0.01;
  const maxCdReduction = coefficients['max_cd_reduction'] ?? 0.7;
  const costScaling = skill.cost_scaling_factor ?? 1.15;
  const baseCost = skill.base_cost_gold ?? 100;
  const baseCd = skill.base_cooldown_seconds ?? 30;

  return {
    ...skill,
    cost_at_5: Math.round(baseCost * Math.pow(costScaling, 5)),
    cost_at_10: Math.round(baseCost * Math.pow(costScaling, 10)),
    cost_at_25: Math.round(baseCost * Math.pow(costScaling, 25)),
    eff_cd_at_10: Math.round(baseCd * (1 - Math.min(cdReduction * 10, maxCdReduction)) * 100) / 100,
    eff_cd_at_25: Math.round(baseCd * (1 - Math.min(cdReduction * 25, maxCdReduction)) * 100) / 100,
  };
}
```

// Top-level state in GameConfigs.tsx
const [configs, setConfigs] = useState<Record<string, GameConfig>>({});
const [dirtyKeys, setDirtyKeys] = useState<Set<string>>(new Set());

const updateConfig = (key: string, value: any) => {
  setConfigs(prev => ({ ...prev, [key]: { ...prev[key], value_json: value } }));
  setDirtyKeys(prev => new Set(prev).add(key));
};

const saveChanges = async () => {
  for (const key of dirtyKeys) {
    await fetch(`/api/admin/game-configs/${key}`, {
      method: 'PATCH',
      body: JSON.stringify({ value_json: configs[key].value_json }),
    });
  }
  setDirtyKeys(new Set());
};
