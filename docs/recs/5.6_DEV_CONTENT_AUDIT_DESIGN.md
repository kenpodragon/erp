## 1. Architecture Overview

```
Runtime Fallback Paths (existing backend code — instrumented):
  ├── story_mode.py ──── log_content_audit() ──── dev_content_audit table
  ├── audio resolution ── log_content_audit()
  └── boss completion ─── log_content_audit()
```

### 1.2 Shared Patterns

- **Admin auth:** All endpoints use `Depends(get_current_admin)`.
- **Pagination:** `?page=1&page_size=50` with `{"items": [...], "total": N, "page": P, "page_size": S}`.
- **Audit logging:** Status changes logged via `write_audit_log()`.
- **Error responses:** 404 for not found, 422 for validation.

## 2. Shared Logging Helper

### 2.1 `dev_audit_service.py` — `log_content_audit()`

```python
from typing import Optional
from sqlmodel import Session, select, col
from models.story_mode import DevContentAudit
from datetime import datetime, timezone


def log_content_audit(
    session: Session,
    audit_type: str,
    entity_type: str,
    entity_id: Optional[int],
    entity_name: Optional[str],
    missing_field: Optional[str],
    scene_id: Optional[int] = None,
    zone_level: Optional[int] = None,
) -> Optional[DevContentAudit]:
    """Log a content audit record with deduplication.

    Returns the new record if created, None if deduplicated (skipped).
    """
    # Dedup check: skip if matching open/acknowledged/in_progress record exists
    query = select(DevContentAudit).where(
        DevContentAudit.audit_type == audit_type,
        DevContentAudit.status.notin_(["resolved", "wont_fix"]),
    )
    if entity_id is not None:
        query = query.where(DevContentAudit.entity_id == entity_id)
    if missing_field is not None:
        query = query.where(DevContentAudit.missing_field == missing_field)
    if scene_id is not None:
        query = query.where(DevContentAudit.scene_id == scene_id)

    existing = session.exec(query).first()
    if existing:
        return None  # Deduplicated

    record = DevContentAudit(
        audit_type=audit_type,
        entity_type=entity_type,
        entity_id=entity_id,
        entity_name=entity_name,
        missing_field=missing_field,
        scene_id=scene_id,
        zone_level=zone_level,
        status="open",
        logged_at=datetime.now(timezone.utc),
    )
    session.add(record)
    # Note: caller is responsible for session.commit() or session.flush()
    return record
```

### 3.1 `GET /api/admin/dev-audit`

**Query Params:** `?audit_type=missing_stat&status=open&entity_type=enemy&search=golem&date_from=2026-03-01&date_to=2026-03-15&page=1&page_size=50`

All params optional. Default: `status` excludes "resolved" and "wont_fix" (shows open + acknowledged + in_progress).

**Response:**

```json
      "id": 1,
      "audit_type": "missing_stat",
      "entity_type": "enemy",
      "entity_id": 42,
      "entity_name": "Shadow Wraith",
      "missing_field": "entity_gameplay_data",
      "scene_id": 15,
      "scene_title": "The Dark Corridor",
      "zone_level": 3,
      "status": "open",
      "logged_at": "2026-03-14T10:30:00Z"
    }

  "total": 17,
  "page": 1,
  "page_size": 50
```

**Service Logic:**
```python
def list_audit_records(
    session: Session,
    audit_type: Optional[str] = None,
    status: Optional[str] = None,
    entity_type: Optional[str] = None,
    search: Optional[str] = None,
    date_from: Optional[datetime] = None,
    date_to: Optional[datetime] = None,
    page: int = 1,
    page_size: int = 50,
    exclude_resolved: bool = True,
) -> dict:
    query = select(DevContentAudit)

    if audit_type:
        query = query.where(DevContentAudit.audit_type == audit_type)
    if status:
        query = query.where(DevContentAudit.status == status)
    elif exclude_resolved:
        query = query.where(DevContentAudit.status.notin_(["resolved", "wont_fix"]))
    if entity_type:
        query = query.where(DevContentAudit.entity_type == entity_type)
    if search:
        query = query.where(col(DevContentAudit.entity_name).ilike(f"%{search}%"))
    if date_from:
        query = query.where(DevContentAudit.logged_at >= date_from)
    if date_to:
        query = query.where(DevContentAudit.logged_at <= date_to)

    total = session.exec(select(func.count()).select_from(query.subquery())).one()

    records = session.exec(
        query.order_by(DevContentAudit.logged_at.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
    ).all()

    # Enrich with scene titles
    scene_ids = [r.scene_id for r in records if r.scene_id]
    scene_titles = {}
    if scene_ids:
        scenes = session.exec(
            select(Scene.id, Scene.title).where(col(Scene.id).in_(scene_ids))
        ).all()
        scene_titles = {s.id: s.title for s in scenes}

    items = []
    for r in records:
        d = _audit_to_dict(r)
        d["scene_title"] = scene_titles.get(r.scene_id) if r.scene_id else None
        items.append(d)

    return {"items": items, "total": total, "page": page, "page_size": page_size}
```

```python
def get_audit_summary(session: Session) -> dict:
    status_counts = dict(session.exec(
        select(DevContentAudit.status, func.count())
        .group_by(DevContentAudit.status)
    ).all())

    type_counts = dict(session.exec(
        select(DevContentAudit.audit_type, func.count())
        .where(DevContentAudit.status.notin_(["resolved", "wont_fix"]))
        .group_by(DevContentAudit.audit_type)
    ).all())

    all_statuses = ["open", "acknowledged", "in_progress", "resolved", "wont_fix"]
    by_status = {s: status_counts.get(s, 0) for s in all_statuses}

    return {
        "by_status": by_status,
        "by_type": type_counts,
        "total": sum(by_status.values()),
    }
```

### 3.4 `PATCH /api/admin/dev-audit/{id}`

**Request Body:**
```json
{
  "status": "acknowledged"
}
```

**Validation:**
- `status`: must be one of `"open"`, `"acknowledged"`, `"in_progress"`, `"resolved"`, `"wont_fix"`.

**Service Logic:**
```python
VALID_STATUSES = {"open", "acknowledged", "in_progress", "resolved", "wont_fix"}

def update_audit_status(session: Session, record_id: int, status: str) -> dict:
    if status not in VALID_STATUSES:
        raise HTTPException(422, f"Invalid status: {status}")

    record = session.get(DevContentAudit, record_id)
    if not record:
        raise HTTPException(404, "Audit record not found")

    record.status = status
    session.add(record)
    session.commit()
    return _audit_to_dict(record)
```

### 3.5 `POST /api/admin/dev-audit/bulk-status`

**Request Body:**
```json
{
  "record_ids": [1, 2, 3, 5],
  "status": "resolved"
}
```

**Response:**
```json
{
  "updated_count": 4
}
```

**Service Logic:**
```python
def bulk_update_status(session: Session, record_ids: list[int], status: str) -> dict:
    if status not in VALID_STATUSES:
        raise HTTPException(422, f"Invalid status: {status}")

    updated = 0
    for rid in record_ids:
        record = session.get(DevContentAudit, rid)
        if record:
            record.status = status
            session.add(record)
            updated += 1

    session.commit()
    return {"updated_count": updated}
```

### 4.6 Deep-Link Fix Routing

```typescript
// In AuditTable.tsx — "Fix →" button click handler
function getFixRoute(record: AuditRecord): string | null {
  switch (record.audit_type) {
    case 'missing_stat':
    case 'missing_sprite':
    case 'missing_sfx':
      return `/admin/world-builder?tab=narrative&entity_id=${record.entity_id}`;
    case 'missing_entity':
      return `/admin/world-builder?tab=narrative&scene_id=${record.scene_id}`;
    case 'missing_atmosphere':
      return `/admin/world-builder?tab=narrative&chapter_id=${record.entity_id}`;
    case 'missing_lore_text':
      if (record.entity_type === 'book') {
        return `/admin/world-builder?tab=narrative&book_id=${record.entity_id}`;
      }
      return `/admin/world-builder?tab=narrative&chapter_id=${record.entity_id}`;
    case 'generic_stat_block':
      return `/admin/world-builder?tab=classification&sub=audit&entity_id=${record.entity_id}`;
    default:
      // Generic fallback: link to entity editor if entity_id present
      return record.entity_id
        ? `/admin/world-builder?tab=narrative&entity_id=${record.entity_id}`
        : null;
  }
}
```

## 6. Design Decisions (Resolved)

| # | Decision | Resolution |
|---|----------|------------|
| D1 | Backend organization | Single `admin_dev_audit.py` route file + `dev_audit_service.py` service |
| D2 | Dedup implementation | Application-level check in `log_content_audit()` before insert |
| D3 | Filter options | Dynamic from data (`SELECT DISTINCT`) — no hardcoded type list |
| D4 | Default status filter | Show "active" records (open + acknowledged + in_progress) by default |
| D5 | Fix routing | Client-side switch on `audit_type` → URL with query params |
