### 1.2 Shared Patterns

- **Concurrency model:** Last-write-wins. No optimistic locking.
- **Audit logging:** All mutations call `write_audit_log(session, admin_email, action, target_type, target_id, details, ip)`.
- **Admin auth:** All endpoints use `Depends(get_current_admin)`.
- **Pagination:** List endpoints accept `?page=1&page_size=50` and return `{"items": [...], "total": N, "page": P, "page_size": S}`.
- **Deletion blocking:** Delete endpoints check for child/FK references and return 409.
- **Error responses:** 422 for validation, 404 for not found, 409 for constraint violations.

## 2. Visual Weight Editor

### 2.1 Backend: `PATCH /api/admin/classification/visual-behaviors/{id}` (Extended)

The existing visual behavior PATCH endpoint (5.3, in `admin_classification.py`) is extended to accept the new `stat_weights` field. No new endpoint needed.

**Extended Request Body (new field):**
```json
{
  "stat_weights": {
    "size": {"strength": 0.6, "agility": 0.2, "intelligence": 0.2},
    "speed": {"strength": 0.1, "agility": 0.7, "intelligence": 0.2},
    "vfx_intensity": {"strength": 0.1, "agility": 0.1, "intelligence": 0.8},
    "clamps": {"size": [0.5, 2.5], "speed": [0.3, 1.5], "vfx_intensity": [0.0, 1.0]}
  }
}
```

**Validation (service layer):**
```python
def validate_stat_weights(stat_weights: dict) -> None:
    PROPERTIES = ["size", "speed", "vfx_intensity"]
    STATS = ["strength", "agility", "intelligence"]

    for prop in PROPERTIES:
        if prop in stat_weights:
            for stat in STATS:
                val = stat_weights[prop].get(stat, 0)
                if not (0.0 <= val <= 1.0):
                    raise HTTPException(422, f"{prop}.{stat} must be between 0.0 and 1.0")

    if "clamps" in stat_weights:
        for prop in PROPERTIES:
            if prop in stat_weights["clamps"]:
                clamp = stat_weights["clamps"][prop]
                if not (isinstance(clamp, list) and len(clamp) == 2):
                    raise HTTPException(422, f"clamps.{prop} must be [min, max]")
                if clamp[0] > clamp[1]:
                    raise HTTPException(422, f"clamps.{prop} min must be <= max")
```

### 3.8 Backend: `POST /api/admin/scaling/wave-presets/{id}/apply-to-scenes`

Bulk-writes the preset's config into `scene_wave_configs` for all scenes in assigned books/chapters that lack a scene-level config.

**Request Body:**

```python
def apply_preset_to_scenes(session: Session, preset_id: int, target: str, target_id: int, overwrite: bool) -> dict:
    preset = session.get(WavePreset, preset_id)
    if not preset:
        raise HTTPException(404, "Wave preset not found")

    # Get target scenes
    if target == "book":
        scene_ids = session.exec(
            select(Scene.id)
            .join(Chapter, Scene.chapter_id == Chapter.id)
            .where(Chapter.book_id == target_id)
        ).all()
    else:  # chapter
        scene_ids = session.exec(
            select(Scene.id).where(Scene.chapter_id == target_id)
        ).all()

    created = 0
    skipped = 0
    for sid in scene_ids:
        existing = session.exec(
            select(SceneWaveConfig).where(SceneWaveConfig.scene_id == sid)
        ).first()

        if existing and not overwrite:
            skipped += 1
            continue

        if existing and overwrite:
            for key, val in preset.config.items():
                if key != "spawn_pattern":  # spawn_pattern not in scene_wave_configs
                    setattr(existing, key, val)
            existing.updated_at = _now()
            session.add(existing)
        else:
            swc = SceneWaveConfig(
                scene_id=sid,
                max_enemies_per_wave=preset.config.get("max_enemies_per_wave", 5),
                wave_count=preset.config.get("wave_count", 10),
                spawn_interval_ms=preset.config.get("spawn_interval_ms", 2000),
                scaling_factor=preset.config.get("scaling_factor", 1.0),
                hp_multiplier=preset.config.get("hp_multiplier", 1.0),
                gold_multiplier=preset.config.get("gold_multiplier", 1.0),
            )
            session.add(swc)
            created += 1

    session.commit()
    return {"created_count": created, "skipped_count": skipped, "skipped_reason": "existing scene_wave_configs"}
```

## 5. Scaling Preview

**Data Sources (fetched on mount):**

### 5.2 Computation: `scaling-utils.ts`

```typescript
interface ScalingProjection {
  chapter_number: number;
  chapter_title: string;
  base_hp: number;
  base_gold: number;
  wave_count: number;
  enemies_per_wave: number;
  spawn_interval_ms: number;
  effective_difficulty: number; // composite index
}

interface ScalingConfig {
  game_configs: Record<string, any>;
  curve: CurveData | null;
  wave_preset: WavePresetConfig | null;
}

function computeProjections(
  config: ScalingConfig,
  chapters: Chapter[],
  book_chapter_offset: number
): ScalingProjection[] {
  const hpFactor = config.game_configs["hp_scaling_factor"] ?? 1.15;
  const goldMult = config.game_configs["session_gold_multiplier"] ?? 1.0;

  return chapters.map((ch, i) => {
    const pos = i + 1; // chapter position within book
    const posStr = String(pos);

    // Get curve multipliers (fallback to last defined or 1.0)
    const curveEntry = getCurveEntry(config.curve, posStr);

    // Get wave config (from preset or defaults)
    const waveConfig = config.wave_preset?.config ?? {
      max_enemies_per_wave: config.game_configs["wave_default_max_enemies"] ?? 5,
      wave_count: config.game_configs["wave_default_wave_count"] ?? 10,
      spawn_interval_ms: config.game_configs["wave_default_spawn_interval_ms"] ?? 2000,
    };

    const base_hp = Math.round(Math.pow(hpFactor, pos) * curveEntry.hp);
    const base_gold = Math.round(10 * goldMult * curveEntry.gold); // base 10 gold
    const enemies = Math.round(waveConfig.max_enemies_per_wave * curveEntry.wave_density);
    const spawn_ms = Math.round(waveConfig.spawn_interval_ms * curveEntry.spawn_speed);
    const wave_count = waveConfig.wave_count;

    // Composite difficulty index: hp * density / spawn_speed
    const effective = base_hp * enemies / Math.max(spawn_ms / 1000, 0.2);

    return {
      chapter_number: pos,
      chapter_title: ch.title,
      base_hp, base_gold, wave_count,
      enemies_per_wave: enemies,
      spawn_interval_ms: spawn_ms,
      effective_difficulty: Math.round(effective),
    };
  });
}

function getCurveEntry(curve: CurveData | null, pos: string): CurveMultipliers {
  if (!curve) return { hp: 1, gold: 1, wave_density: 1, spawn_speed: 1 };

  if (curve[pos]) return curve[pos];

  // Fallback to last defined entry
  const keys = Object.keys(curve).map(Number).sort((a, b) => a - b);
  const lastKey = keys[keys.length - 1];
  return curve[String(lastKey)] ?? { hp: 1, gold: 1, wave_density: 1, spawn_speed: 1 };
}
```

**Service Logic:**
```python
PRESET_CATEGORIES = ["combat", "upgrades"]
PRESET_EXTRA_KEYS = ["session_gold_multiplier", "first_clear_multiplier"]

def capture_current_config(session: Session) -> dict:
    # Get all game_configs in target categories
    configs = session.exec(
        select(GameConfig).where(
            or_(
                col(GameConfig.category).in_(PRESET_CATEGORIES),
                col(GameConfig.key).in_(PRESET_EXTRA_KEYS)
            )
        )
    ).all()

    snapshot = {c.key: c.value_json for c in configs}

    # Find currently active curve (from default or first book's assignment)
    default_curve = session.exec(
        select(DifficultyCurve).where(DifficultyCurve.is_default == True)
    ).first()

    default_wave = session.exec(
        select(WavePreset).where(WavePreset.is_default == True)
    ).first()

    return {
        "config_snapshot": snapshot,
        "captured_categories": PRESET_CATEGORIES,
        "extra_keys": PRESET_EXTRA_KEYS,
        "total_keys": len(snapshot),
        "current_active_curve_id": default_curve.id if default_curve else None,
        "current_active_wave_preset_id": default_wave.id if default_wave else None,
    }
```

### 6.5 Backend: `POST /api/admin/scaling/presets/{id}/apply`

Atomically applies a preset: writes config_snapshot to game_configs, assigns curve to books, assigns wave preset.

**Response:**
```json
{
  "configs_updated": 26,
  "curve_assigned_to_books": 3,
  "wave_preset_assigned": true,
  "previous_active_preset": "Experimental v2"
}
```

**Service Logic:**
```python
def apply_preset(session: Session, preset_id: int, admin_email: str) -> dict:
    preset = session.get(DifficultyPreset, preset_id)
    if not preset:
        raise HTTPException(404, "Preset not found")

    # 1. Write config_snapshot values to game_configs
    configs_updated = 0
    for key, value in preset.config_snapshot.items():
        gc = session.exec(
            select(GameConfig).where(GameConfig.key == key)
        ).first()
        if gc:
            gc.value_json = value
            gc.updated_at = _now()
            session.add(gc)
            configs_updated += 1

    # 2. Assign difficulty curve to all books (if preset has one)
    books_updated = 0
    if preset.difficulty_curve_id:
        books = session.exec(select(Book)).all()
        for book in books:
            book.difficulty_curve_id = preset.difficulty_curve_id
            session.add(book)
            books_updated += 1

    # 3. Set wave preset as default (if preset has one)
    wave_assigned = False
    if preset.wave_preset_id:
        # Clear existing defaults
        session.exec(
            update(WavePreset).where(WavePreset.is_default == True)
            .values(is_default=False)
        )
        wave = session.get(WavePreset, preset.wave_preset_id)
        if wave:
            wave.is_default = True
            session.add(wave)
            wave_assigned = True

    # 4. Update active preset
    session.exec(
        update(DifficultyPreset).where(DifficultyPreset.is_active == True)
        .values(is_active=False)
    )
    previous_active = session.exec(
        select(DifficultyPreset).where(DifficultyPreset.is_active == True)
    ).first()
    preset.is_active = True
    session.add(preset)

    session.commit()

    write_audit_log(session, admin_email, "apply_preset", "difficulty_preset", preset_id,
                    {"configs_updated": configs_updated, "books_updated": books_updated}, None)

    return {
        "configs_updated": configs_updated,
        "curve_assigned_to_books": books_updated,
        "wave_preset_assigned": wave_assigned,
        "previous_active_preset": previous_active.name if previous_active else None,
    }
```
